\section{Judgments}

We work with three mutually defined judgment forms:
\begin{align*}
  &\wf{\Gamma}                  &&\text{$\Gamma$ is a well-formed context}\\
  &\hastype{\Gamma}{M}{A}       &&\text{$M$ has type $A$ in context $\Gamma$}\\
  &\conveq{\Gamma}{M}{N}{A}     &&\text{$M$ and $N$ are definitionally equal at type $A$}
\end{align*}
Contexts are snoc-lists: $\Gamma ::= \cdot \mid \Gamma, x{:}A$.
We write $M\sigma$ for the result of applying a substitution
$\sigma$ to a term~$M$, and $[t]$ for the substitution
$\Gamma \to \Gamma, x{:}A$ extending the identity by~$t$,
so that $B[a]$ means $B(\mathrm{id}, a)$.

  \and
  \inferrule[\textsc{Beta}]

     \hastype{\Gamma}{a}{A}}

       {\substone{B}{a}}}
  \and
  \inferrule[\textsc{Pi-Cong}]
    {\conveq{\Gamma}{A}{A'}{\U} \\
     \conveq{\Gamma, x{:}A}{B}{B'}{\U}}
    {\conveq{\Gamma}
       {\Pitype{x{:}A}{B}}
       {\Pitype{x{:}A'}{B'}}
       {\U}}
\end{mathpar}
\noindent
Note that in \textsc{Pi-Cong} the codomain premise compares $B$ and
$B'$ in the context extended by the \emph{left} domain~$A$, not
by~$A'$.  This is the standard convention: since $A = A' : \U$ holds,
one could equivalently use~$A'$ (via the conversion rule on contexts),
but fixing the left side avoids a mutual dependency between the
congruence rule and context conversion.

We write $M^+$ for the weakening of a term~$M$ by one fresh
variable, i.e.\ $M^+$ is $M$ viewed in the extended context
$\Gamma, x{:}A$ where it does not depend on~$x$.

  \inferrule[\textsc{Funext}]
    {\hastype{\Gamma}{A}{\U} \\
     \conveq{\Gamma, x{:}A}{\App{\wkn{f}}{x}}{\App{\wkn{g}}{x}}{B} \\
     \hastype{\Gamma}{f}{\Pitype{x{:}A}{B}} \\
     \hastype{\Gamma}{g}{\Pitype{x{:}A}{B}}}
    {\conveq{\Gamma}{f}{g}{\Pitype{x{:}A}{B}}}
  \and
  \inferrule[\textsc{App-Fun}]
    {\hastype{\Gamma}{A}{\U} \\
     \hastype{\Gamma, x{:}A}{B}{\U} \\
     \conveq{\Gamma}{f}{f'}{\Pitype{x{:}A}{B}} \\
     \hastype{\Gamma}{a}{A}}
    {\conveq{\Gamma}
       {\App{f}{a}}
       {\App{f'}{a}}
       {\substone{B}{a}}}
  \and
  \inferrule[\textsc{App-Arg}]
    {\hastype{\Gamma}{A}{\U} \\
     \hastype{\Gamma, x{:}A}{B}{\U} \\
     \hastype{\Gamma}{f}{\Pitype{x{:}A}{B}} \\
     \conveq{\Gamma}{a}{a'}{A}}
    {\conveq{\Gamma}
       {\App{f}{a}}
       {\App{f}{a'}}
       {\substone{B}{a}}}
\end{mathpar}

\begin{theorem}[Eta-conversion]
  If\/ $\hastype{\Gamma}{A}{\U}$, $\hastype{\Gamma, x{:}A}{B}{\U}$,
  and $\hastype{\Gamma}{f}{\Pitype{x{:}A}{B}}$, then
  \[
    \conveq{\Gamma}{f}{\Lamterm{x{:}A}{\App{\wkn{f}}{x}}}{\Pitype{x{:}A}{B}}.
  \]
\end{theorem}

\begin{proof}
  By \textsc{Funext} it suffices to show
  $\conveq{\Gamma, x{:}A}{\App{\wkn{f}}{x}}
    {\App{\wkn{(\Lamterm{x{:}A}{\App{\wkn{f}}{x}})}}{x}}{B}$.
  Since weakening distributes under $\lambda$, we have
  $\wkn{(\Lamterm{x{:}A}{M})} = \Lamterm{x{:}\wkn{A}}{\wkn{M}}$
  (where the inner weakening acts on the next-to-last variable,
  skipping the binder).  The right-hand side is therefore a
  $\beta$-redex.  Applying \textsc{Beta} and the substitution
  identity $\substone{\wkn{M}}{x_0} = M$ gives the result.
  Formalised in \texttt{EtaConversion.agda}.
\end{proof}

\section{Head reduction}

We define head reduction $M \leadsto N$ as the compatible closure
of $\beta$-reduction at the head:
\begin{mathpar}
  \inferrule[\textsc{HR-Beta}]
    { }
    {\App{(\Lamterm{x{:}A}{M})}{a} \leadsto \substone{M}{a}}
  \and
  \inferrule[\textsc{HR-App}]
    {M \leadsto M'}
    {\App{M}{a} \leadsto \App{M'}{a}}
\end{mathpar}

We write $M \Red N$ for the reflexive-transitive closure.
The validity relation uses a wrapper $\mathsf{Red}\;\Gamma\;M\;N\;A$,
which carries only a proof of $M \Red N$; the context~$\Gamma$ and
type~$A$ are phantom parameters kept for inference convenience.
In~\cite{CH2018}, the reduction relation is defined as a subrelation
of conversion, requiring both $M \Red N$ and $\conveq{\Gamma}{M}{N}{A}$.
Here, reduction is purely syntactic: the conversion witnesses are
instead stored directly at the leaves of the bundled validity
relation ($\mathsf{Val2}$/$\mathsf{EqVal2}$), so the
$\mathsf{Red}$ wrapper need not carry them.

\section{Finite elements and the information ordering}

Following Coquand--Huber~\cite{CH2018}, we interpret types and terms
as ideal-domain elements, approximated from below by \emph{finite
elements}.  The set $\FinEl$ and its auxiliary $\FinFun$ are defined
by mutual induction:
\[
  \begin{array}{rcl}
    u,v,a,b &::=& \Bot \mid \UCode \mid \FunEl{g} \mid \PiC{a}{f} \\[3pt]
    g,f &\in& \FinFun \;=\; \mathsf{List}(\FinEl \times \FinEl)
  \end{array}
\]
Here $\Bot$ is the least element (no information),
$\UCode$ is the code for the universe,
$\FunEl{g}$ is a function element with graph~$g$
(a finite list of input--output pairs),
and $\PiC{a}{f}$ is a code for a $\Pi$-type with domain
approximation~$a$ and codomain graph~$f$.

\paragraph{Binary supremum.}
A partial binary supremum $u \Sup v$ is defined by case analysis:
it combines compatible codes ($\PiC{a}{f} \Sup \PiC{b}{g} =
\PiC{a \Sup b}{f \mathbin{+\!\!+} g}$,
$\FunEl{g} \Sup \FunEl{h} = \FunEl{g \mathbin{+\!\!+} h}$)
and returns $\Bot$ for incompatible pairs
(e.g.\ $\UCode \Sup \FunEl{g} = \Bot$).

\paragraph{Evaluation of finite functions.}
Given a finite function $f = [(u_1,v_1), \dots, (u_k,v_k)]$ and an
input $u$, the sup-based evaluation collects all outputs whose keys
are below~$u$:
\[
  \EvalFun(f, u) \;=\; \bigsqcup\{ v_i \mid u_i \LeCode u \}.
\]

\paragraph{Information ordering.}
The ordering $u \LeCode v$ on $\FinEl$ is defined mutually with
an ordering on finite functions.  The key clauses are:
\begin{align*}
  \Bot \LeCode v &\;=\; \top \\
  \UCode \LeCode \UCode &\;=\; \top \\
  \FunEl{g} \LeCode \FunEl{h}
    &\;\Longleftrightarrow\; g \LeCode_{\mathsf{fun}} h \\
  \PiC{a}{f} \LeCode \PiC{b}{g}
    &\;\Longleftrightarrow\;
      a \LeCode b \;\wedge\; f \LeCode_{\mathsf{fun}} g
\end{align*}
where the ordering on finite functions uses the
\emph{finite condition}:
\[
  g \LeCode_{\mathsf{fun}} h
  \quad\Longleftrightarrow\quad
  \forall (u_i, v_i) \in g.\;
    v_i \LeCode \EvalFun(h, u_i).
\]
All other combinations (e.g.\ $\UCode \LeCode \FunEl{g}$)
are empty.

\paragraph{Coherence.}
A finite element is \emph{coherent} ($\Coh(u)$) when its
internal data is consistent.  Specifically,
$\Coh(\PiC{a}{f})$ requires $\Coh(a)$ and that $f$ is a
\emph{coherent finite function}: whenever two edges
$(u_1,v_1),(u_2,v_2) \in f$ have compatible keys
($\mathsf{Comp}(u_1,u_2)$), their values are also compatible
($\mathsf{Comp}(v_1,v_2)$).  This condition ensures that finite
functions represent well-defined continuous maps between domains.

\paragraph{Finite membership.}
The \emph{finite typing} relation $u \FinMem a$ (``$u$ is an element
of the type coded by~$a$'') is defined by mutual recursion on
$u$ and~$a$:
\begin{align*}
  \Bot \FinMem a &\;\Longleftrightarrow\; a \FinMem \UCode
    \qquad\text{(every type has $\Bot$)} \\
  \UCode \FinMem \UCode &\;=\; \top \\
  \FunEl{g} \FinMem \PiC{a}{f}
    &\;\Longleftrightarrow\;
    \mathsf{FinMemFun}(g,a,f) \;\wedge\;
    \Coh(\FunEl{g}) \;\wedge\;
    \PiC{a}{f} \FinMem \UCode \\
  \PiC{a}{f} \FinMem \UCode
    &\;\Longleftrightarrow\;
    a \FinMem \UCode \;\wedge\;
    \mathsf{FinMemAllU}(f,a) \;\wedge\;
    \mathsf{CohFunTail}(f)
\end{align*}
where $\mathsf{FinMemFun}(g,a,f)$ checks that every edge
$(u_i,v_i) \in g$ satisfies $u_i \FinMem a$ and
$v_i \FinMem \EvalFun(f, u_i)$,
and $\mathsf{FinMemAllU}(f,a)$ checks that every edge
$(a_i,b_i) \in f$ satisfies $a_i \FinMem a$ and $b_i \FinMem \UCode$.

Key structural properties:
\begin{itemize}
  \item $u \FinMem a$ implies $\Coh(u)$, $\Coh(a)$, and $a \FinMem \UCode$.
  \item Membership is monotone: if $u \FinMem a$ and $a \LeCode a'$
    with $a' \FinMem \UCode$, then $u \FinMem a'$.
\end{itemize}

\section{Selections}

A \emph{selection} of a finite function $f$ is a compatible subset
of edges from~$f$, yielding an aggregate key~$u$ and aggregate
value~$v$.  Formally, $\mathsf{Selection}(f, u, v)$ is an inductive
type with three constructors:
\begin{mathpar}
  \inferrule[\textsc{Sel-Nil}]
    { }
    {\mathsf{Selection}(\mathsf{nil}, \Bot, \Bot)}
  \and
  \inferrule[\textsc{Sel-Skip}]
    {\mathsf{Selection}(g, u, v)}
    {\mathsf{Selection}((a_i,b_i){::}g, \; u, \; v)}
  \and
  \inferrule[\textsc{Sel-Take}]
    {\mathsf{Comp}(a_i, u) \\
     \mathsf{Comp}(b_i, v) \\
     \mathsf{Selection}(g, u, v)}
    {\mathsf{Selection}((a_i,b_i){::}g, \;
       a_i \Sup u, \; b_i \Sup v)}
\end{mathpar}
The compatibility conditions in \textsc{Sel-Take} ensure that the
accumulated $u$ and $v$ remain coherent.  A key property is that
$\mathsf{Selection}(f,u,v)$ implies $v \LeCode \EvalFun(f,u)$.

Why selections rather than individual edges?  The evaluation
$\EvalFun(f,u)$ is defined as a supremum: it aggregates the outputs
of \emph{all} edges in~$f$ whose keys lie below~$u$.  Working with
single edges would not capture this aggregate character.  A selection
encodes a finite compatible subfamily of edges whose joins approximate
$\EvalFun(f,u)$ from below, and the compatibility check guarantees
that these joins are well-defined (i.e.\ coherent).  This is the
mechanism that lets the relational semantics and the validity relation
handle $\Pi$-types and $\lambda$-terms in a purely finite way, without
ever computing an infinite supremum.

\section{Relational semantics on finite approximants}

\begin{definition}[Evaluation relation]
  Given a raw term $M$ in context of size~$n$, a finite environment
  $\rho$ (mapping each variable to a $\FinEl$), and a finite
  element~$u$, the relation $\EvalRel{M}{\rho}{u}$ (``$M$ evaluates
  to at least~$u$ in~$\rho$'') is defined by recursion on~$M$:

  \begin{itemize}
    \item \textbf{Variable:}\;
      $\EvalRel{x_i}{\rho}{u}$
      iff $\Coh(u)$ and $u \LeCode \rho(i)$.

    \item \textbf{Universe:}\;
      $\EvalRel{\U}{\rho}{u}$ iff $\Coh(u)$ and $u \LeCode \UCode$.

    \item \textbf{Application:}\;
      $\EvalRel{\App{M}{N}}{\rho}{\Bot}$ always holds.
      For non-$\Bot$ output~$u$: there exists~$v$ such that
      $\EvalRel{N}{\rho}{v}$ and
      $\EvalRel{M}{\rho}{\FunEl{[(v,u)]}}$.
      (One-edge form: the function $M$ maps input~$v$
      to output~$u$.)

    \item \textbf{Lambda:}\;
      $\EvalRel{\Lamterm{x{:}A}{M}}{\rho}{\Bot}$ always holds.
      $\EvalRel{\Lamterm{x{:}A}{M}}{\rho}{\FunEl{g}}$
      requires a domain code~$a$ with
      $\Coh(\FunEl{g})$, $a \FinMem \UCode$,
      $\EvalRel{A}{\rho}{a}$,
      and for every selection of $g$ with key~$u$ and value~$v$,
      there exists~$x \LeCode u$ with $x \FinMem a$ and
      $\EvalRel{M}{\rho[x]}{v}$.

    \item \textbf{Pi:}\;
      $\EvalRel{\Pitype{x{:}A}{B}}{\rho}{\Bot}$ always holds.
      $\EvalRel{\Pitype{x{:}A}{B}}{\rho}{\PiC{a}{f}}$
      requires $\Coh(\PiC{a}{f})$,
      $\EvalRel{A}{\rho}{a}$,
      and a witness $a'$ with $\EvalRel{A}{\rho}{a'}$ such that
      for every selection of $f$ with key~$u$ and value~$v$,
      there exists~$x \LeCode u$ with $x \FinMem a'$ and
      $\EvalRel{B}{\rho[x]}{v}$.
  \end{itemize}
\end{definition}

The relation is purely finite and has no reference to ideal
semantic functions.  Key properties:
\begin{itemize}
  \item \emph{Coherence extraction}: $\EvalRel{M}{\rho}{u}$
    implies $\Coh(u)$.
  \item \emph{Compatibility}: if $\EvalRel{M}{\rho}{u}$ and
    $\EvalRel{M}{\rho}{v}$ with $\Coh(\rho)$, then
    $\mathsf{Comp}(u,v)$.
  \item \emph{Monotonicity in $\rho$}: if $\rho \LeCode \rho'$
    pointwise and $\EvalRel{M}{\rho}{u}$, then
    $\EvalRel{M}{\rho'}{u}$.
  \item \emph{Downward closure}: $\EvalRel{M}{\rho}{u}$ and
    $u' \LeCode u$ with $\Coh(u')$ implies
    $\EvalRel{M}{\rho}{u'}$.
  \item \emph{Sup closure}: $\EvalRel{M}{\rho}{u}$ and
    $\EvalRel{M}{\rho}{v}$ implies
    $\EvalRel{M}{\rho}{u \Sup v}$ (when the environment is
    coherent).
\end{itemize}

\section{Soundness of the relational semantics}

\begin{definition}[Fitting environment]
  An environment $\rho$ \emph{fits} a context $\Gamma$ (written
  $\mathsf{Fits}(\Gamma, \rho)$) when, for each variable
  $x_i : A_i$ in~$\Gamma$, there exists a code~$a'$ such that
  $\rho(i) \FinMem a'$ and $\EvalRel{A_i}{\rho{\upharpoonright i}}{a'}$.
\end{definition}

\begin{definition}[Typed evaluation]
  $\mathsf{Typed}(M,A,\rho,u)$ asserts that $u$ can be enlarged to
  some $u' \geq u$ which is a member of some code~$a'$ that
  $A$ evaluates to: there exist $u'$ and $a'$ such that
  $u \LeCode u'$, $\EvalRel{M}{\rho}{u'}$, $u' \FinMem a'$, and
  $\EvalRel{A}{\rho}{a'}$.
\end{definition}

\begin{theorem}[Typing soundness --- Theorem~1 of~\cite{CH2018}]
  \label{thm:typing-sound}
  If\/ $\hastype{\Gamma}{M}{A}$ and $\mathsf{Fits}(\Gamma, \rho)$,
  then for every $u$ with $\EvalRel{M}{\rho}{u}$,
  we have $\mathsf{Typed}(M,A,\rho,u)$.
\end{theorem}

\begin{theorem}[Conversion soundness]
  \label{thm:conv-sound}
  If\/ $\conveq{\Gamma}{M}{N}{A}$ and $\mathsf{Fits}(\Gamma,\rho)$,
  then:
  \begin{enumerate}
    \item Typing: $\mathsf{Typed}(M,A,\rho,u)$ for all
      $\EvalRel{M}{\rho}{u}$, and likewise for~$N$.
    \item Bidirectional transfer: for all~$u$,
      $\EvalRel{M}{\rho}{u} \Longleftrightarrow \EvalRel{N}{\rho}{u}$.
  \end{enumerate}
\end{theorem}

The proofs proceed by mutual induction on the typing and conversion
derivations.  The $\beta$ and function extensionality cases use the
evaluation--substitution lemma: if $\EvalRel{M}{\rho[v]}{u}$ and
$\EvalRel{N}{\rho}{v}$, then
$\EvalRel{M[N/x]}{\rho}{u}$ (and conversely).
Formalised in \texttt{TypingSemantics.agda} and
\texttt{LemmaForTS.agda}, with 0~postulates.

\section{Logical relation (validity)}

The validity relation is defined by mutual structural recursion on
the type code~$a \in \FinEl$.  It connects the syntactic judgments
to the finite semantics.

\begin{definition}[Validity]
  For a context~$\Gamma$, terms $M$, $A$, a realizer $u$, and a
  type code~$a$, we define mutually:

  \medskip
  \noindent\textbf{Term validity} $\ValR(\Gamma, M, A, u, a)$
  by case split on~$a$:
  \begin{itemize}
    \item At $a = \Bot$: always $\top$ (trivially valid).
    \item At $a = \UCode$: reduces to type validity
      $\mathsf{ValTy}(\Gamma, M, u)$.
    \item At $a = \PiC{b}{f}$ with $u = \FunEl{g}$: a
      $\Sigma$-type containing
      \begin{enumerate}
        \item type validity of $A$ at $\PiC{b}{f}$
          (head-reduces to some $\Pi\,A_0\,B_0$),
        \item $\Coh(\FunEl{g})$ and
          $\mathsf{FinMemFun}(g,b,f)$,
        \item for every selection of $g$ with key~$u'$ and
          value~$v$, and argument $N$ with
          $\ValR(\Gamma, N, A_0, u', b)$:
          \[
            \ValR(\Gamma, \App{M}{N}, B_0[N], v, \EvalFun(f, u')).
          \]
      \end{enumerate}
    \item At $a = \PiC{b}{f}$ with $u \neq \FunEl{g}$: always $\top$.
  \end{itemize}

  \medskip
  \noindent\textbf{Type validity} $\mathsf{ValTy}(\Gamma, M, u)$
  at $u = \PiC{b}{f}$:
  a $\Sigma$-type containing terms $A$, $B$ such that
  $M \Red \Pitype{x{:}A}{B}$ in~$\Gamma$,
  $\mathsf{CohFunTail}(f)$,
  $\mathsf{FinMemAllU}(f,b)$,
  type validity $\mathsf{ValTy}(\Gamma, A, b)$,
  and for every selection of $f$ with key~$u'$ and value~$v$,
  and argument~$N$ with $\ValR(\Gamma, N, A, u', b)$:
  $\mathsf{ValTy}(\Gamma, B[N], v)$.

  \medskip
  \noindent\textbf{Equality validity} $\EqValR(\Gamma, M, N, A, u, a)$
  bundles validity of both $M$ and $N$, plus type-equality data at
  $\Pi$-codes.
\end{definition}

The key feature of this logical relation is that it recurses on
the type code~$a$ rather than on syntactic types.  Since
$\EvalFun(f, u')$ is structurally smaller than $\PiC{b}{f}$
(its rank is bounded by $\mathsf{rk}_{\mathsf{fun}}(f)
< \mathsf{rk}(\PiC{b}{f})$), the recursion is well-founded
despite the type-in-type universe.

\paragraph{Monotonicity properties.}
Three mutual monotonicity families are proved:
\begin{itemize}
  \item \emph{Downward in $a$}: if $a_0 \LeCode a_1$,
    $u \FinMem a_0$, $\Coh(a_0)$, and $a_1 \FinMem \UCode$, then
    $\ValR(\Gamma,M,A,u,a_1) \Rightarrow \ValR(\Gamma,M,A,u,a_0)$.
  \item \emph{Upward in $a$}: if $a_0 \LeCode a_1$,
    $u \FinMem a_0$, $u \FinMem a_1$, $\Coh(a_0)$, $\Coh(a_1)$,
    and $\mathsf{ValTy}(\Gamma, A, a_1)$, then
    $\ValR(\Gamma,M,A,u,a_0) \Rightarrow \ValR(\Gamma,M,A,u,a_1)$.
  \item \emph{Restriction in $u$}: if $u' \LeCode u$,
    $u' \FinMem a$, and $u \FinMem a$, then
    $\ValR(\Gamma,M,A,u,a) \Rightarrow \ValR(\Gamma,M,A,u',a)$.
\end{itemize}
Formalised in \texttt{Validity.agda} with 0~postulates.

\section{Adequacy}

The adequacy theorem is the bridge between the syntactic typing
judgments and the semantic validity relation.  It is proved using
the two-substitution technique of~\cite{CH2018}, Theorem~2.

\begin{definition}[Bundled validity]
  $\mathsf{Val2}(\Gamma, M, A, u, a)$ refines $\ValR$ by
  requiring $\hastype{\Gamma}{M}{A}$ at the leaves (at
  $\Bot$-realizer or $\Bot$-code positions).  Similarly,
  $\mathsf{EqVal2}$ requires $\conveq{\Gamma}{M}{N}{A}$.
  These bundled versions are defined in \texttt{Validity2.agda}.
\end{definition}

\begin{theorem}[Adequacy --- two-substitution form]
  \label{thm:adequacy}
  The following are proved by mutual induction on typing/conversion
  derivations, instantiated at two substitutions $\sigma_1$,
  $\sigma_2$ simultaneously:
  \begin{enumerate}
    \item If\/ $\hastype{\Gamma}{M}{A}$, then for all codes
      $u,a$ with $\EvalRel{M}{\rho}{u}$ and
      $\EvalRel{A}{\rho}{a}$:
      \[
        \mathsf{Val2}(\Delta,\, M[\sigma],\, A[\sigma],\, u,\, a).
      \]
    \item If\/ $\conveq{\Gamma}{M}{N}{A}$, then under the same
      conditions:
      \[
        \mathsf{EqVal2}(\Delta,\, M[\sigma],\, N[\sigma],\,
          A[\sigma],\, u,\, a).
      \]
  \end{enumerate}
\end{theorem}

The proof uses the substitution lemma (well-typed substitutions
preserve typing and conversion), context conversion, the soundness
theorems~\ref{thm:typing-sound} and~\ref{thm:conv-sound},
and the monotonicity properties of the validity relation.
Key cases:
\begin{itemize}
  \item \textsc{Beta}: uses the evaluation--substitution
    correspondence and $\beta$-expansion/contraction lemmas for
    $\mathsf{Val2}$.
  \item \textsc{Funext}: uses weakening for
    $\mathsf{EvalRel}$ and the selection-based Pi clauses.
  \item \textsc{Conv}: uses the bidirectional transfer from
    conversion soundness and the downward/upward monotonicity
    of $\mathsf{Val2}$.
  \item \textsc{Pi}: constructs a $\PiC{b}{f}$-realizer by
    building selections from the domain and codomain adequacy
    hypotheses.
\end{itemize}
Formalised in \texttt{Adequacy2.agda} with 0~postulates.

\section{Main result: Pi injectivity}

\begin{theorem}[Corollary 6; Coquand--Huber 2018, p.\,661]
  If\/ $\conveq{\Gamma}{A_0}{\Pitype{x{:}B_1}{F_1}}{\U}$,
  then there exist $B_0$, $F_0$ such that
  \begin{enumerate}
    \item $A_0 \leadsto^* \Pitype{x{:}B_0}{F_0}$,
    \item $\conveq{\Gamma}{B_0}{B_1}{\U}$,
    \item $\conveq{\Gamma, x{:}B_0}{F_0}{F_1}{\U}$.
  \end{enumerate}
\end{theorem}

\begin{corollary}[Pi--Pi injectivity]
  If\/ $\conveq{\Gamma}{\Pitype{x{:}A_0}{B_0}}{\Pitype{x{:}A_1}{B_1}}{\U}$,
  then
  \[
    \conveq{\Gamma}{A_0}{A_1}{\U}
    \qquad\text{and}\qquad
    \conveq{\Gamma, x{:}A_0}{B_0}{B_1}{\U}.
  \]
\end{corollary}

\begin{proof}[Proof outline (Theorem)]
  The proof proceeds by domain-theoretic adequacy.
  \begin{enumerate}
    \item From $\conveq{\Gamma}{A_0}{\Pi B_1\,F_1}{\U}$, extract
      $\hastype{\Gamma}{A_0}{\U}$ by presupposition
      (\textsc{SubstitutionLemma}).
    \item Apply the soundness theorem (\textsc{convSound'}) at the
      \emph{bottom environment} $\rho_\bot$ (mapping every variable to~$\bot$)
      to obtain a bidirectional evaluation transfer: for every finite
      approximation~$u$,
      \[
        \mathsf{EvalRel}(\Pi B_1\,F_1, \rho_\bot, u)
        \;\Longrightarrow\;
        \mathsf{EvalRel}(A_0, \rho_\bot, u).
      \]
    \item Instantiate with $u = \mathsf{PiCode}(\bot, \mathsf{nil})$,
      which is coherent because
      $\mathsf{CoherentFunTail}(\mathsf{nil}) = \top$.
      The evaluation $\mathsf{EvalRel}(\Pi B_1\,F_1, \rho_\bot,
      \mathsf{PiCode}(\bot,\mathsf{nil}))$ is constructible trivially
      (domain~$\bot$, empty function graph, vacuous selection body).
      Transfer gives $\mathsf{EvalRel}(A_0, \rho_\bot,
      \mathsf{PiCode}(\bot,\mathsf{nil}))$.
    \item Apply the bundled equality adequacy theorem
      (\textsc{adequacyEqSub2}) with the identity substitution
      at~$\rho_\bot$ to obtain
      \[
        \mathsf{EqVal2}\;\Gamma\;A_0\;(\Pi B_1\,F_1)\;\U\;
        (\mathsf{PiCode}(\bot,\mathsf{nil}))\;\mathsf{UCode}.
      \]
    \item This unfolds to $\mathsf{EqValTyPi2}$, a $\Sigma$-type
      containing:
      \begin{itemize}
        \item $\mathsf{Red}\;\Gamma\;A_0\;(\Pi\,B_0\,F_0)\;\U$
          \quad (giving part~1),
        \item $\conveq{\Gamma}{B_0}{B_1'}{\U}$ and
          $\conveq{\Gamma, x{:}B_0}{F_0}{F_1'}{\U}$,
        \item $\mathsf{Red}\;\Gamma\;(\Pi B_1\,F_1)\;(\Pi B_1'\,F_1')\;\U$.
      \end{itemize}
      Since $\Pi$ is a head normal form,
      $\mathsf{Red\text{-}unique\text{-}Pi}$ forces $B_1' = B_1$ and
      $F_1' = F_1$, giving parts~2 and~3.
  \end{enumerate}
\end{proof}

\begin{proof}[Proof of Corollary]
  Apply the theorem to
  $\conveq{\Gamma}{\Pi A_0\,B_0}{\Pi A_1\,B_1}{\U}$.
  This yields $B_0'$, $F_0'$ with
  $\Pi A_0\,B_0 \leadsto^* \Pi B_0'\,F_0'$.
  Since $\Pi$ is a head normal form, $\mathsf{Red\text{-}unique\text{-}Pi}$
  gives $B_0' = A_0$ and $F_0' = B_0$, so the domain and codomain
  conversions become
  $\conveq{\Gamma}{A_0}{A_1}{\U}$ and
  $\conveq{\Gamma, x{:}A_0}{B_0}{B_1}{\U}$.
\end{proof}

\medskip\noindent\textbf{Key intermediate lemmas.}
The proof above relies on several results that deserve explicit mention:
\begin{itemize}
  \item \emph{Presupposition}
    (\texttt{SubstitutionLemma.agda}): from a conversion judgment
    $\conveq{\Gamma}{M}{N}{A}$ one can extract
    $\hastype{\Gamma}{M}{A}$, $\hastype{\Gamma}{N}{A}$, and
    $\hastype{\Gamma}{A}{\U}$.  This is proved by mutual induction
    on typing and conversion, using the substitution and renaming
    lemmas.
  \item \emph{Conversion soundness} (\texttt{convSound'} in
    \texttt{TypingSemantics.agda}): if
    $\conveq{\Gamma}{M}{N}{A}$ and $\mathsf{EvalRel}(N,\rho,u)$,
    then $\mathsf{EvalRel}(M,\rho,u)$ (and vice versa).
  \item \emph{Bundled adequacy} (\texttt{adequacyEqSub2} in
    \texttt{Adequacy2.agda}): the bridge from $\mathsf{EvalRel}$
    to the validity relation $\mathsf{EqVal2}$, which packages both
    a semantic value and the corresponding syntactic typing/conversion
    derivation.
  \item \emph{Reduction uniqueness for} $\Pi$
    (\texttt{Red-unique-Pi}): since $\Pi$ is a head normal form,
    $\Pi\,A\,B \leadsto^* \Pi\,A'\,B'$ implies $A = A'$ and $B = B'$.
    This is the step that turns the ``up to reduction'' output of
    adequacy into exact equalities on the domain and codomain.
\end{itemize}

\medskip\noindent
The entire proof is formalised in Agda with \texttt{--without-K
--exact-split}, no \texttt{--type-in-type}, and no postulates
(in the files on which \texttt{PiInjectivity.agda} depends).
