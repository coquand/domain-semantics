\section{Overview}

We describe the extension of the logical relation (validity) for dependent
type theory with $\Type:\Type$ to handle $\Sigma$-types.  The presentation
follows Coquand--Huber (2018), where the logical relation is indexed by
\emph{codes} from a domain-theoretic model.  The key new difficulty compared
to $\Pi$-types is in the head-reduction transport lemma.

\section{Codes and projections}

The set $\FinEl$ of codes includes:
\[
  u,v ::= \Bot \mid \Type \mid \Prop \mid \lambda g \mid \Pi\,b\,f
         \mid \Sigma\,b\,f \mid (u,v)
\]
A code $\Sigma\,b\,f$ represents a $\Sigma$-type whose domain has code~$b$
and whose codomain is the finite function~$f$.
A code $(u,v)$ represents a pair value.

We use a uniform projection notation for both terms and codes:
\[
  \pj{M}{1},\quad \pj{M}{2} \qquad\text{(first and second projection of a term)}
\]
\[
  \pj{w}{1},\quad \pj{w}{2} \qquad\text{(first and second projection of a code)}
\]
On codes, the projections are defined by:
\[
  \pj{(u,v)}{1} = u, \qquad \pj{w}{1} = \Bot \text{ otherwise}
\]
\[
  \pj{(u,v)}{2} = v, \qquad \pj{w}{2} = \Bot \text{ otherwise}
\]

\section{Validity at $\Sigma$-types}

\begin{definition}[Term validity at $\Sigma$]
We define $\Gamma \HasType M : T \mid w : \Sigma\,b\,f$ to hold when:
\begin{enumerate}
\item $\ValTy\;\Gamma\;T\;(\Sigma\,b\,f)$ (type validity of~$T$), which
  includes $T \HeadRed \Sigma\,A\,B$ with $\Gamma \HasType A : \Type$ and
  $\Gamma,x\!:\!A \HasType B : \Type$, together with the edge functions;
\item $\Gamma \HasType \pj{M}{1} : A \mid \pj{w}{1} : b$ (validity of the
  first projection);
\item $\Gamma \HasType \pj{M}{2} : B[\pj{M}{1}] \mid \pj{w}{2} : f(\pj{w}{1})$
  (validity of the second projection).
\end{enumerate}
\end{definition}

\begin{remark}
When $w$ is not a pair $(u,v)$, we have $\pj{w}{1} = \Bot$ and
$\pj{w}{2} = \Bot$.  Since $\Val$ at code $\Bot$ is trivially $\top$,
conditions (2) and (3) are automatically satisfied.  This means there is no
need for separate case analysis on whether $w$ is a pair or not: the
definition is \emph{uniform} in~$w$.
\end{remark}

\begin{remark}
Compared to the $\Pi$-case, where term validity at $\Pi\,b\,f$ is
defined for $w = \lambda g$ and stores
$\Val\;\Gamma\;(\App\,M\,N)\;(B[N])\;v\;(f(u))$, the type $B[N]$ does not
mention~$M$.  For $\Sigma$, the second component's type $B[\pj{M}{1}]$
\emph{does} mention~$M$.  This is the source of the difficulty in
head-reduction transport.
\end{remark}

\section{Type validity at $\Sigma$}

\begin{definition}[Type validity at $\Sigma$]
$\ValTy\;\Gamma\;T\;(\Sigma\,b\,f)$ holds when:
\begin{itemize}
\item $T \HeadRed \Sigma\,A\,B$ and $\Gamma \HasType T \ConvTm \Sigma\,A\,B : \Type$;
\item $\Gamma \HasType A : \Type$ and $\Gamma,x\!:\!A \HasType B : \Type$;
\item $\ValTy\;\Gamma\;A\;b$ (domain type validity);
\item For each $(u,v) \in f$ and $N$ with $\Gamma \HasType N : A$ and
  $\Val\;\Gamma\;N\;A\;u\;b$, we have $\ValTy\;\Gamma\;B[N]\;v$ (edge
  function for values);
\item Similarly for the equality edge.
\end{itemize}
\end{definition}

\section{Type-equality transport}

\begin{lemma}[Transport along type equality]\label{lem:transport}
For all codes $u$ and $a$:
\[
  \EqValTy\;\Gamma\;C\;C'\;a \;\;\Longrightarrow\;\;
  \Val\;\Gamma\;N\;C\;u\;a \;\;\Longrightarrow\;\;
  \Val\;\Gamma\;N\;C'\;u\;a
\]
\end{lemma}

\begin{proof}
By induction on $a$.  The type parameter $C$ appears in $\Val\;\Gamma\;N\;C\;u\;a$
only in two cases: $a = \Pi\,b\,f$ with $u = \lambda g$, and
$a = \Sigma\,b\,f$ with $u = (u',v')$.  In these cases the validity
record stores a head reduction $C \HeadRed \Pi\,A\,B$
or $C \HeadRed \Sigma\,A\,B$, and $\EqValTy\;\Gamma\;C\;C'\;a$
provides the corresponding data for~$C'$, allowing
the record to be rebuilt.  In all other cases---including $a = \Type$,
where $\Val$ depends on~$N$ but not on~$C$---the result is immediate.
\end{proof}

\begin{remark}
This lemma plays a central role in the $\Sigma$ case of
head-reduction transport (Lemma~\ref{lem:headred}, Step~4), where it
transports $\Val$ from type $B[\pj{M'}{1}]$ to type $B[\pj{M}{1}]$.
For $\Pi$-types alone, this lemma is not needed in head-reduction transport.
\end{remark}

\section{Head-reduction transport}

The central lemma for adequacy is:

\begin{lemma}[Head-reduction produces equality]\label{lem:headred}
Given:
\begin{itemize}
\item $M \HeadRed M'$ and $\Gamma \HasType M \ConvTm M' : T$,
\item $\Gamma \HasType M' : T \mid u : a$ \quad (validity of the reduct),
\end{itemize}
we can conclude:
\[
  \Gamma \HasType M' = M : T \mid u : a
\]
which bundles $\Val\;\Gamma\;M\;T\;u\;a$ and $\Val\;\Gamma\;M'\;T\;u\;a$
and the equality $\EqVal\;\Gamma\;M'\;M\;T\;u\;a$.
\end{lemma}

\begin{remark}
The return type is $\EqVal$ (not just $\Val$).  This is essential
for the $\Sigma$ case: the equality at the first projection is needed to
transport the second projection's type.
\end{remark}

\subsection{The $\Pi$ case}

For $a = \Pi\,b\,f$ and $u = \lambda g$, the validity data for~$M'$
stores, for each $(u_0, v_0) \in g$ and argument~$N$:
\[
  \Val\;\Gamma\;(\App\,M'\,N)\;(B_0[N])\;v_0\;(f(u_0))
\]
Since $M \HeadRed M'$, we have $\App\,M\,N \HeadRed \App\,M'\,N$.
The type $B_0[N]$ does not mention~$M$, so the induction hypothesis
applies directly at code $(v_0, f(u_0))$:
\[
  \EqVal\;\Gamma\;(\App\,M'\,N)\;(\App\,M\,N)\;(B_0[N])\;v_0\;(f(u_0))
\]
No type transport is needed.

\subsection{The $\Sigma$ case}\label{sec:sigma}

For $a = \Sigma\,b\,f$, we work with the uniform definition from
Section~3.  The validity data for~$M'$ includes:
\begin{align}
  &\Val\;\Gamma\;\pj{M'}{1}\;A\;\pj{w}{1}\;b  \label{eq:fstM'}\\
  &\Val\;\Gamma\;\pj{M'}{2}\;B[\pj{M'}{1}]\;\pj{w}{2}\;f(\pj{w}{1})  \label{eq:sndM'}
\end{align}

We need to produce the same data for~$M$, plus equality.

\medskip
\noindent\textbf{Step 1: First projection.}
From $M \HeadRed M'$ we get $\pj{M}{1} \HeadRed \pj{M'}{1}$.  From
$\Gamma \HasType M \ConvTm M' : \Sigma\,A\,B$ we derive (by the congruence
rule for projections):
\[
  \Gamma \HasType \pj{M}{1} \ConvTm \pj{M'}{1} : A
\]
Together with~\eqref{eq:fstM'}, the induction hypothesis at code
$(\pj{w}{1}, b)$ gives:
\begin{equation}\label{eq:eqfst}
  \EqVal\;\Gamma\;\pj{M'}{1}\;\pj{M}{1}\;A\;\pj{w}{1}\;b
\end{equation}
This bundles $\Val\;\Gamma\;\pj{M}{1}\;A\;\pj{w}{1}\;b$.

\medskip
\noindent\textbf{Step 2: Type transport via the edge function.}
The equality~\eqref{eq:eqfst} and the equality edge function from $\ValTy$ at
$\Sigma\,b\,f$ give:
\begin{equation}\label{eq:eqty}
  \EqValTy\;\Gamma\;B[\pj{M'}{1}]\;B[\pj{M}{1}]\;f(\pj{w}{1})
\end{equation}
This is the key step: the \emph{equality} edge function converts the equality of
\emph{values} ($\pj{M'}{1} = \pj{M}{1}$) into an equality of \emph{types}
($B[\pj{M'}{1}] = B[\pj{M}{1}]$).  This step has no analogue in the $\Pi$ case.

Note that the \emph{value} edge (which only takes $\Val$ data) would not suffice
here: we genuinely need the \emph{equality} at the first projection to produce
the type equality for the transport.  This is why
Lemma~\ref{lem:headred} must return $\EqVal$ rather than just $\Val$.

\medskip
\noindent\textbf{Step 3: Second projection.}
From $M \HeadRed M'$ we get $\pj{M}{2} \HeadRed \pj{M'}{2}$.  The congruence
rule for the second projection, applied to $\Gamma \HasType M' \ConvTm M :
\Sigma\,A\,B$ (note the swap, so that $\pj{M'}{1}$ appears in the type), gives:
\[
  \Gamma \HasType \pj{M'}{2} \ConvTm \pj{M}{2} : B[\pj{M'}{1}]
\]
Together with~\eqref{eq:sndM'}, the induction hypothesis at code
$(\pj{w}{2}, f(\pj{w}{1}))$ gives:
\begin{equation}\label{eq:eqsnd}
  \EqVal\;\Gamma\;\pj{M'}{2}\;\pj{M}{2}\;B[\pj{M'}{1}]\;\pj{w}{2}\;f(\pj{w}{1})
\end{equation}
In particular, $\Val\;\Gamma\;\pj{M}{2}\;B[\pj{M'}{1}]\;\pj{w}{2}\;f(\pj{w}{1})$.

\medskip
\noindent\textbf{Step 4: Transport.}
The validity data for~$M$ at $\Sigma$ requires:
\[
  \Val\;\Gamma\;\pj{M}{2}\;B[\pj{M}{1}]\;\pj{w}{2}\;f(\pj{w}{1})
\]
We have $\Val$ at $B[\pj{M'}{1}]$ from Step~3, and $\EqValTy$ between
$B[\pj{M'}{1}]$ and $B[\pj{M}{1}]$ from Step~2.
Lemma~\ref{lem:transport} gives
$\Val\;\Gamma\;\pj{M}{2}\;B[\pj{M}{1}]\;\pj{w}{2}\;f(\pj{w}{1})$. \qed

\section{Modular structure}

The proof has the following modular structure:

\begin{enumerate}
\item \textbf{Domain operations} ($\pj{w}{1}$, $\pj{w}{2}$, $\EvalFun$):
  Pure operations on codes, no typing.

\item \textbf{Validity definition}: Uniform in~$w$ using $\pj{w}{1}$ and
  $\pj{w}{2}$.  No case split on whether $w$ is a pair.

\item \textbf{Transport lemmas} (sym, trans, fwd, restrict, up, down, Sup):
  Each proved by induction on codes.  The $\Sigma$ cases follow the same
  pattern as $\Pi$ because the definition is uniform.

\item \textbf{Head-reduction transport} (Lemma~\ref{lem:headred}):
  Returns $\EqVal$ (not just $\Val$).  The 4-step chain in
  Section~\ref{sec:sigma} uses:
  \begin{itemize}
  \item the induction hypothesis at the first projection (Step~1),
  \item the edge function to get type equality (Step~2),
  \item the induction hypothesis at the second projection (Step~3),
  \item the standard $\EqValTy$-transport (Step~4).
  \end{itemize}

\item \textbf{Adequacy}: Uses head-reduction transport for the
  $\beta$-reduction cases (application, $\pj{}{1}$-$\beta$, $\pj{}{2}$-$\beta$).
\end{enumerate}

\section{Bundling requirements}

For $\Pi$, $\Type$, and $\Prop$ alone, the following simpler choices suffice:
\begin{itemize}
\item Head reduction ($\mathsf{Red}$) stores only $T \HeadRed \Pi\,A\,B$,
  without the convertibility judgment $\Gamma \HasType T \ConvTm \Pi\,A\,B : \Type$.
\item The head-reduction transport lemma can return just $\Val$
  (validity of the expanded term), without the equality $\EqVal$.
\item Term validity $\Val\;\Gamma\;M\;T\;u\;a$ need not be bundled with
  type validity $\ValTy\;\Gamma\;T\;a$.
\end{itemize}
For $\Sigma$-types, all three must be strengthened:
\begin{enumerate}
\item \textbf{$\mathsf{Red}$ must bundle convertibility.}
  The head-reduction transport for $\Sigma$ needs to derive
  $\Gamma \HasType \pj{M}{1} \ConvTm \pj{M'}{1} : A$ (via the congruence
  rule for projections).  This congruence rule requires
  $\Gamma \HasType M \ConvTm M' : \Sigma\,A\,B$, which is obtained from
  the input $\Gamma \HasType M \ConvTm M' : T$ by converting the type using
  $\Gamma \HasType T \ConvTm \Sigma\,A\,B : \Type$---the convertibility
  stored in the type validity record alongside the head reduction
  $T \HeadRed \Sigma\,A\,B$.

\item \textbf{Head-reduction transport must return $\EqVal$.}
  The equality $\EqVal$ at the first projection is needed to invoke
  the equality edge function (Step~2), which produces the type equality
  $\EqValTy\;\Gamma\;B[\pj{M'}{1}]\;B[\pj{M}{1}]\;f(\pj{w}{1})$ used
  in the transport (Step~4).

\item \textbf{$\Val$ must be bundled with $\ValTy$.}
  Step~2 uses the equality edge function from $\ValTy\;\Gamma\;T\;(\Sigma\,b\,f)$.
  This must be accessible from within the term validity data.
  Therefore $\Val\;\Gamma\;M\;T\;w\;(\Sigma\,b\,f)$ must include
  $\ValTy\;\Gamma\;T\;(\Sigma\,b\,f)$ as a component.
\end{enumerate}
